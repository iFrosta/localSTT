cmake_minimum_required(VERSION 3.20)
project(LocalSTTNative LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "Build the complete application on Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
    add_compile_options(/utf-8 /EHsc /W3)
endif()
# One version for both applications: localstt/__init__.py is the source, and the
# update check compares it with the newest published release.
set(LOCALSTT_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../localstt/__init__.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LOCALSTT_VERSION_FILE}")
file(STRINGS "${LOCALSTT_VERSION_FILE}" LOCALSTT_VERSION_LINE REGEX "^__version__ *= *\"[0-9]+\\.[0-9]+\\.[0-9]+\"")
if(NOT LOCALSTT_VERSION_LINE MATCHES "\"([0-9]+)\\.([0-9]+)\\.([0-9]+)\"")
    message(FATAL_ERROR "No __version__ = \"X.Y.Z\" in ${LOCALSTT_VERSION_FILE}")
endif()
set(LOCALSTT_VERSION_MAJOR ${CMAKE_MATCH_1})
set(LOCALSTT_VERSION_MINOR ${CMAKE_MATCH_2})
set(LOCALSTT_VERSION_PATCH ${CMAKE_MATCH_3})
set(LOCALSTT_VERSION "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
# A generated header rather than /D flags: the resource compiler needs the version as
# a string literal, and quotes do not survive every generator's escaping.
configure_file(version.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/version.h" @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
message(STATUS "LocalSTT version ${LOCALSTT_VERSION}")
set(LOCALSTT_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../native-install" CACHE PATH "Where the Release executables are written")
add_library(localstt_core common.cpp platform.cpp commands.cpp hotkeys.cpp dictionary.cpp updates.cpp)
target_include_directories(localstt_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
target_link_libraries(localstt_core PUBLIC winmm mfplat mfreadwrite mfuuid ole32 user32 shell32 propsys advapi32 uuid winhttp)
add_library(localstt_app engine.cpp ct2.cpp tokenizer.cpp features.cpp ui.cpp)
set(CT2_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ctranslate2/include" CACHE PATH "CTranslate2 headers")
set(CT2_IMPORT_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ctranslate2/ctranslate2.lib" CACHE FILEPATH "CTranslate2 import library")
target_include_directories(localstt_app PRIVATE ${CT2_INCLUDE_DIR})
target_link_libraries(localstt_app PUBLIC ${CT2_IMPORT_LIBRARY})
target_link_libraries(localstt_app PUBLIC localstt_core localstt_live ws2_32 crypt32 comctl32 dwmapi uxtheme gdi32)
add_executable(LocalSTT WIN32 main.cpp app.rc)
add_executable(localstt-cli main.cpp app.rc)
foreach(target LocalSTT localstt-cli)
    target_link_libraries(${target} PRIVATE localstt_app)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${LOCALSTT_OUTPUT_DIR}")
endforeach()
add_library(localstt_live live_dictation.cpp)
target_compile_features(localstt_live PUBLIC cxx_std_20)
target_include_directories(localstt_live PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
    add_compile_options(/utf-8)
    target_compile_options(localstt_live PRIVATE /utf-8)
endif()
include(CTest)
set(LOCALSTT_TEST_MODEL "large-v3-turbo" CACHE STRING "Bundled CTranslate2 model for tokenizer verification")
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(test_localstt_live tests/live_dictation.cpp)
    target_link_libraries(test_localstt_live PRIVATE localstt_live Threads::Threads)
    add_test(NAME live_dictation COMMAND test_localstt_live)
    add_executable(test_localstt_commands tests/commands.cpp)
    target_link_libraries(test_localstt_commands PRIVATE localstt_core)
    add_test(NAME commands COMMAND test_localstt_commands)
    add_executable(test_localstt_platform tests/platform.cpp)
    target_link_libraries(test_localstt_platform PRIVATE localstt_core)
    add_test(NAME platform COMMAND test_localstt_platform)
    add_executable(test_localstt_hotkeys tests/hotkeys.cpp)
    target_link_libraries(test_localstt_hotkeys PRIVATE localstt_core)
    add_test(NAME hotkeys COMMAND test_localstt_hotkeys)
    add_executable(test_localstt_updates tests/updates.cpp)
    target_link_libraries(test_localstt_updates PRIVATE localstt_core)
    add_test(NAME updates COMMAND test_localstt_updates)
    add_executable(test_localstt_dictionary tests/dictionary.cpp)
    target_link_libraries(test_localstt_dictionary PRIVATE localstt_core)
    add_test(NAME dictionary COMMAND test_localstt_dictionary "${CMAKE_CURRENT_SOURCE_DIR}/..")
    add_executable(test_localstt_features tests/features.cpp features.cpp)
    target_include_directories(test_localstt_features PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME features COMMAND test_localstt_features)
    add_executable(test_localstt_tokenizer tests/tokenizer.cpp tokenizer.cpp)
    target_link_libraries(test_localstt_tokenizer PRIVATE localstt_core)
    add_test(NAME tokenizer COMMAND test_localstt_tokenizer "${CMAKE_CURRENT_SOURCE_DIR}/../native-install/models/ctranslate2/${LOCALSTT_TEST_MODEL}/tokenizer.json" "${CMAKE_CURRENT_SOURCE_DIR}/tests/tokenizer-reference.json")
    add_executable(test_localstt_ct2 EXCLUDE_FROM_ALL tests/ct2.cpp)
    target_link_libraries(test_localstt_ct2 PRIVATE localstt_app)
    set_target_properties(test_localstt_ct2 PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${LOCALSTT_OUTPUT_DIR}")
endif()
